1//! Cuora impl of [`cuora_secret::Secrets`] — the org's unified
2//! secrets-broker adapter (P-043; absorbs `hypha-secret-cuora`,
3//! `peregrin-secret-cuora`, `bupha-secret-cuora`, `mimus-cuora`).
4//!
5//! Each secret NAME resolves to a fixed cuora ref under the
6//! consumer's scope: `CuoraSecrets::new("bupha")` maps
7//! `GITHUB_TOKEN` → `cuora://bupha/GITHUB_TOKEN`. Values are fetched
8//! through cuora so audit events, guardrails, and vault policy apply
9//! exactly as for every other consumer; the host never stores a
10//! value itself.
11//!
12//! Ordered fallback scopes compose via
13//! [`CuoraSecrets::with_fallbacks`] — `get` walks primary then each
14//! fallback (own-scope-first: a name present under several scopes
15//! keeps resolving to the primary's value), `set` always writes the
16//! primary. This covers the ex-`peregrin-secret-cuora`
17//! `peregrin → bupha → secrets-ci-webhooks` chain and the
18//! ex-`bupha-secret-cuora` `bupha → secrets-ci-webhooks` chain —
19//! the scope lists stay consumer policy, the walk is port
20//! machinery. [`CuoraSecrets::get_many`] is the batched form for
21//! listeners resolving hundreds of names at startup.
22//!
23//! Two transports, selected per store via [`CuoraTransport`]:
24//!
25//! - `cli` (default) — shell `cuora get`/`put` per call. Secret
26//!   resolution happens a handful of times per run/verb, so a
27//!   per-call process is cheap and needs no shared mutable state.
28//! - `serve` — a persistent `cuora serve` child speaking the
29//!   ndjson JSON-RPC wire (`cuora-serve/1`) on stdio — see
30//!   `serve`/[`CuoraServe`]. Spawned lazily on first use, reused
31//!   for the client's lifetime, respawned on death. This is the
32//!   transport `cuora serve` exists for (orchestrator embedding).
33//!
34//! A missing `cuora` binary, a denied op, or an absent ref all
35//! answer `None` — "not here" — so the caller's chain falls through
36//! to the legacy stores (OS keyring, env file) exactly like a
37//! keyring miss on a headless box, on either transport.
38//!
39//! ## Env-var convention
40//!
41//! The shared crate never reads the environment itself — consumers
42//! pass their knob *names* to [`CuoraSecrets::from_env`] /
43//! [`CuoraTransport::from_env`]. The convention is
44//! `<PRODUCT>_CUORA_BIN` for the binary override (tests stub it)
45//! and `<PRODUCT>_CUORA_TRANSPORT` for `cli`/`serve` — e.g.
46//! `PEREGRIN_CUORA_BIN` + `PEREGRIN_CUORA_TRANSPORT`. `CUORA_*` and
47//! `--config`/`--env` resolution stay cuora's own — children inherit
48//! the environment, so `CUORA_ENV` picks the vault env with no
49//! consumer-side plumbing (the serve child resolves env at spawn;
50//! per-request `env` overrides are not used).
51//!
52//! ## Process-wide default
53//!
54//! Worth sharing under the serve transport: `CuoraSecrets` is cheap
55//! to construct but the serve child's whole point is persistence —
56//! a fresh store per lookup would spawn and reap `cuora serve` on
57//! every secret read, worse than the CLI. Consumers that want a
58//! shared default keep their own `OnceLock` (the scope/env names
59//! are theirs); under `cli` sharing is a no-op behaviorally.

72/// `Secrets` over cuora — CLI or `cuora serve` transport per
73/// [`CuoraTransport`]. Generic over the ref `scope` (the
74/// `cuora://<scope>/…` namespace) and the audit `actor` (defaults to
75/// the scope — every org consumer's actor name equals its scope).
76///
77/// The serve leg holds one shared child per `CuoraSecrets` clone
78/// group ([`Arc`]); clones share the process.
79#[derive(Clone, Debug)]
80pub struct CuoraSecrets {
81    /// The `cuora` binary — the consumer's `*_CUORA_BIN` override
82    /// when set, else `cuora` resolved on PATH.
83    bin: PathBuf,
84    /// Ref namespace — every secret name lives under
85    /// `cuora://<scope>/…` so `cuora list` shows the whole surface.
86    scope: String,
87    /// Audit actor — `--actor` on the CLI leg, `actor` on the wire
88    /// for serve (a claimed label, like every local actor claim).
89    actor: String,
90    /// CLI vs persistent serve transport.
91    transport: CuoraTransport,
92    /// The serve client — constructed eagerly but spawns lazily on
93    /// first request, so the CLI transport never pays for it.
94    serve: Arc<CuoraServe>,
95    /// Ordered fallback scopes — `get` walks these after the
96    /// primary misses (see [`Self::with_fallbacks`]). Each carries
97    /// its own serve child under the serve transport.
98    fallbacks: Vec<CuoraSecrets>,
99}

101impl CuoraSecrets {
102    /// A store over scope `scope` with the `cuora` binary on PATH,
103    /// actor = scope, and the CLI transport. Use
104    /// [`Self::from_env`] to honor the consumer's env knobs, or the
105    /// `with_*` builders for explicit values (tests).
106    #[must_use]
107    pub fn new(scope: &str) -> Self {
108        Self {
109            bin: PathBuf::from("cuora"),
110            scope: scope.to_string(),
111            actor: scope.to_string(),
112            transport: CuoraTransport::Cli,
113            serve: Arc::new(CuoraServe::new(PathBuf::from("cuora"), scope)),
114            fallbacks: Vec::new(),
115        }
116    }

118    /// The env-selected store — `bin_env` names the variable holding
119    /// the `cuora` binary override (unset → `cuora` on PATH),
120    /// `transport_env` the `<PRODUCT>_CUORA_TRANSPORT` knob. See the
121    /// crate docs for the convention.
122    #[must_use]
123    pub fn from_env(scope: &str, bin_env: &str, transport_env: &str) -> Self {
124        let bin = std::env::var_os(bin_env).map_or_else(|| PathBuf::from("cuora"), PathBuf::from);
125        Self::new(scope)
126            .with_bin(bin)
127            .with_transport(CuoraTransport::from_env(transport_env))
128    }

130    /// A store over `bin` — the test seam and the "I know what I'm
131    /// doing" constructor. Transport stays CLI unless
132    /// [`Self::with_transport`] says otherwise. Propagates to
133    /// fallback legs so a stub binary covers the whole chain.
134    #[must_use]
135    pub fn with_bin(mut self, bin: PathBuf) -> Self {
136        self.bin.clone_from(&bin);
137        self.serve = Arc::new(CuoraServe::new(bin.clone(), self.actor.clone()));
138        for f in &mut self.fallbacks {
139            *f = f.clone().with_bin(bin.clone());
140        }
141        self
142    }

144    /// An explicit transport — the test seam and the "I know what
145    /// I'm doing" constructor. Propagates to fallback legs.
146    #[must_use]
147    pub fn with_transport(mut self, transport: CuoraTransport) -> Self {
148        self.transport = transport;
149        for f in &mut self.fallbacks {
150            *f = f.clone().with_transport(transport);
151        }
152        self
153    }

155    /// Override the audit actor (defaults to the scope).
156    /// Propagates to fallback legs — the actor is the *caller*, not
157    /// the scope, when a consumer overrides it.
158    #[must_use]
159    pub fn with_actor(mut self, actor: impl Into<String>) -> Self {
160        self.actor = actor.into();
161        self.serve = Arc::new(CuoraServe::new(self.bin.clone(), self.actor.clone()));
162        for f in &mut self.fallbacks {
163            *f = f.clone().with_actor(self.actor.clone());
164        }
165        self
166    }

168    /// Ordered fallback scopes — `get` tries the primary, then each
169    /// fallback in order; `set` always writes the primary only.
170    ///
171    /// Own-scope-first is deliberate: a name present under several
172    /// scopes keeps resolving to the primary's value, so
173    /// legitimately-divergent scopes are preserved and a fallback
174    /// (e.g. a sealed pack — the migration target) never overrides.
175    /// Typical use is the shared fleet surface:
176    ///
177    /// ```ignore
178    /// CuoraSecrets::from_env("peregrin", BIN_ENV, TRANSPORT_ENV)
179    ///     .with_fallbacks(vec![
180    ///         CuoraSecrets::from_env("bupha", BIN_ENV, TRANSPORT_ENV),
181    ///         CuoraSecrets::from_env("secrets-ci-webhooks", BIN_ENV, TRANSPORT_ENV),
182    ///     ])
183    /// ```
184    #[must_use]
185    pub fn with_fallbacks(mut self, fallbacks: Vec<CuoraSecrets>) -> Self {
186        self.fallbacks = fallbacks;
187        self
188    }

190    /// The fallback legs, in order (empty for a single-scope store).

193        &self.fallbacks

196    /// Batch `get` — resolves every name through the full
197    /// primary+fallback chain, returning one `Option` per name in
198    /// order (`None` = miss, identical to [`Secrets::get`] per
199    /// name).
200    ///
201    /// Sequential `get`s serialize on the wire: under `serve` a
202    /// shared child answers one request at a time and each request
203    /// costs a vault round-trip, so a listener startup resolving
204    /// ~140 names took seconds. `get_many` fans the batch out over
205    /// `workers` parallel lanes — under `serve` each lane is its
206    /// own `cuora serve` child per scope (spawned lazily, reaped
207    /// when the batch returns; audit/guardrail/PDP semantics per
208    /// request are unchanged since every request is an ordinary
209    /// `get`), under `cli` each lane is just a thread since every
210    /// `get` is already its own process.
211    ///
212    /// Secret *values* are never logged; results stay in the
213    /// caller's ctx exactly like `get`.
214    #[must_use]
215    pub fn get_many(&self, names: &[&str], workers: usize) -> Vec<Option<String>> {
216        let workers = workers.min(names.len());
217        if workers <= 1 {
218            return names.iter().map(|name| self.get(name)).collect();
219        }
220        let mut results: Vec<Option<String>> = vec![None; names.len()];
221        std::thread::scope(|scope| {
222            let handles: Vec<_> = (0..workers)
223                .map(|lane| {
224                    scope.spawn(move || {
225                        // Round-robin keeps lanes balanced when a
226                        // few names are slow (remote vault hits).
227                        // Fresh serve children per lane — the
228                        // shared `serve` Arc would serialize the
229                        // whole batch on its request mutex.
230                        let stores = self.lane_stores();
231                        let mut lane_results = Vec::new();
232                        let mut i = lane;
233                        while i < names.len() {
234                            let value = stores.iter().find_map(|s| s.get(names[i]));
235                            lane_results.push((i, value));
236                            i += workers;
237                        }
238                        lane_results
239                    })
240                })
241                .collect();
242            for handle in handles {
243                if let Ok(pairs) = handle.join() {
244                    for (i, value) in pairs {
245                        results[i] = value;
246                    }
247                }
248            }
249        });
250        results
251    }
252
253    /// This store plus its fallbacks rebuilt with fresh serve
254    /// children — one lane's worth of `get_many` workers. Under the
255    /// CLI transport the serve client is never spawned, so the
256    /// rebuild is free.
257    fn lane_stores(&self) -> Vec<CuoraSecrets> {
258        let mut out = Vec::with_capacity(1 + self.fallbacks.len());
259        out.push(self.fresh_serve());
260        out.extend(self.fallbacks.iter().map(CuoraSecrets::fresh_serve));
261        out
262    }
263
264    /// A clone whose serve client is a fresh [`CuoraServe`] — same
265    /// bin/scope/actor/transport/fallbacks shape, but no shared
266    /// child (and no shared *fallbacks'* children either: each leg
267    /// rebuilds recursively).
268    fn fresh_serve(&self) -> Self {
269        Self {
270            bin: self.bin.clone(),
271            scope: self.scope.clone(),
272            actor: self.actor.clone(),
273            transport: self.transport,
274            serve: Arc::new(CuoraServe::new(self.bin.clone(), self.actor.clone())),
275            fallbacks: self
276                .fallbacks
277                .iter()
278                .map(CuoraSecrets::fresh_serve)
279                .collect(),
280        }
281    }

322/// `get` attempts on the CLI transport — transient flakes (T-059:
323/// ~6 empty outputs in ~800 calls) get three tries; the backoff
324/// stays small because the flake clears in milliseconds and the
325/// common path is a single fast local/HTTP read.
326const GET_ATTEMPTS: u32 = 3;
327const GET_RETRY_BACKOFF: Duration = Duration::from_millis(50);
328
329/// The outcome of one `cuora get` invocation, classified for the
330/// retry loop: a definitive answer ends it, anything ambiguous
331/// retries.
332enum GetOnce {
333    /// A value came back — done.
334    Hit(String),
335    /// A definitive "not here" — the engine's not-stored message,
336    /// a guardrail denial, or a usage error. No retry: these
337    /// can't be transient and the fallthrough chain is waiting.
338    Miss,
339    /// Ambiguous — spawn/io failure, empty stdout on a successful
340    /// exit (impossible for a real `put`, which refuses empty
341    /// values), or a nonzero exit that isn't a definitive miss
342    /// (vault transport errors all land here).
343    Retry,
344}
345
346/// Does a failed `cuora get` carry a *definitive* answer — one a
347/// retry can't change? The engine prints `cuora: <msg>` (or
348/// `{"error": …}` under `--json`, which we don't pass) and exits
349/// nonzero for everything, so the message text is the only
350/// signal. The not-stored string is cuora-cli's own miss wording
351/// (`{rref} is not stored in env {env}`); "denied by guardrails"
352/// is its advisory-deny wording; clap usage errors exit 2. Any
353/// drift in those strings degrades to "retry a permanent miss" —
354/// bounded and harmless — never to "retry forever" or a wrong
355/// `Some`.
356fn is_definitive_miss(code: Option<i32>, stderr: &str) -> bool {
357    if code == Some(2) {
358        return true;
359    }
360    stderr.contains("is not stored in env") || stderr.contains("denied by guardrails")
361}

363impl Secrets for CuoraSecrets {
364    fn name(&self) -> &str {
365        "cuora"
366    }
367
368    /// Cheap probe — the `cuora` binary answers `--version`. A
369    /// missing binary means every `get` misses; the chain still
370    /// tries it (this is for `doctor` to report, not to gate on).
371    fn available(&self) -> bool {
372        Command::new(&self.bin)
373            .arg("--version")
374            .stdin(Stdio::null())
375            .stdout(Stdio::null())
376            .stderr(Stdio::null())
377            .status()
378            .map(|s| s.success())
379            .unwrap_or(false)
380    }
381
382    /// Primary scope first, then each fallback in order — a name
383    /// present under several scopes keeps resolving to the primary's
384    /// value (see [`Self::with_fallbacks`]).
385    fn get(&self, name: &str) -> Option<String> {
386        self.get_primary(name)
387            .or_else(|| self.fallbacks.iter().find_map(|f| f.get(name)))
388    }
389
390    /// Deploy-time migration: `cuora put` reads the value on stdin
391    /// (never argv — argv lands in shell history and `/proc`).
392    /// Under serve the value rides the request's `value` field —
393    /// the wire's designed channel, still never argv. Always writes
394    /// the *primary* scope — a shared fleet name's canonical write
395    /// is `cuora put cuora://<canonical-scope>/<NAME>`.
396    fn set(&self, name: &str, value: &str) -> std::io::Result<()> {
397        match self.transport {
398            CuoraTransport::Cli => self.set_cli(name, value),
399            CuoraTransport::Serve => self.serve.put(&self.rref(name), value),
400        }
401    }
402}

404impl CuoraSecrets {
405    /// This leg's own `get` on the selected transport — no
406    /// fallback walk (the trait impl owns that).
407    fn get_primary(&self, name: &str) -> Option<String> {
408        match self.transport {
409            CuoraTransport::Cli => self.get_cli(name),
410            CuoraTransport::Serve => self.serve.get(&self.rref(name)),
411        }
412    }
413
414    /// The `cli` transport's `get` — shell `cuora get` per lookup,
415    /// with a bounded retry for transient failures (T-059).
416    ///
417    /// `cuora get` answers a *definitive* miss on stderr +
418    /// nonzero exit (`… is not stored in env …`) and every other
419    /// failure — vault transport flakes, spawn errors, a wedged
420    /// config load — the same way: empty stdout, nonzero exit.
421    /// Collapsing all of that to `None` in one shot turned a ~1%
422    /// transient flake into silent secret misses, so ambiguous
423    /// outcomes get [`GET_ATTEMPTS`] tries with a short backoff.
424    /// Answers that can't be transient — the not-stored message,
425    /// a guardrail denial, a clap usage error (exit 2) — return
426    /// `None` on the first try so the fallthrough chain doesn't
427    /// pay for them.
428    fn get_cli(&self, name: &str) -> Option<String> {
429        for attempt in 0..GET_ATTEMPTS {
430            if attempt > 0 {
431                std::thread::sleep(GET_RETRY_BACKOFF * attempt);
432            }
433            match self.get_cli_once(name) {
434                GetOnce::Hit(v) => return Some(v),
435                GetOnce::Miss => return None,
436                GetOnce::Retry => {}
437            }
438        }
439        None
440    }
441
442    /// One `cuora get` invocation, classified for [`Self::get_cli`]'s
443    /// retry loop.
444    fn get_cli_once(&self, name: &str) -> GetOnce {
445        let Ok(o) = Command::new(&self.bin)
446            .arg("--actor")
447            .arg(&self.actor)
448            .arg("get")
449            .arg(self.rref(name))
450            .stdin(Stdio::null())
451            .stderr(Stdio::piped())
452            .output()
453        else {
454            // Spawn/io failure (fork pressure, missing binary on a
455            // flaky PATH) — transient by definition.
456            return GetOnce::Retry;
457        };
458        if !o.status.success() {
459            // stderr is small (one `cuora: …` line plus audit
460            // ndjson) — keep enough to classify, cap the read the
461            // pipe already bounded.
462            let err = String::from_utf8_lossy(&o.stderr);
463            return if is_definitive_miss(o.status.code(), &err) {
464                GetOnce::Miss
465            } else {
466                GetOnce::Retry
467            };
468        }
469        // `cuora get` prints the value via println! — strip exactly
470        // one trailing newline (and a CR before it); interior
471        // whitespace is part of the value. `put` refuses empty
472        // secrets, so a successful exit can never legitimately
473        // print nothing — empty stdout is itself the flake.
474        let mut s = String::from_utf8_lossy(&o.stdout).into_owned();
475        if let Some(stripped) = s.strip_suffix('\n') {
476            s = stripped.strip_suffix('\r').unwrap_or(stripped).to_string();
477        }
478        if s.is_empty() {
479            GetOnce::Retry
480        } else {
481            GetOnce::Hit(s)
482        }
483    }
484
485    /// The `cli` transport's `set` — `cuora put` with the value on
486    /// stdin (never argv — argv lands in shell history and `/proc`).
487    fn set_cli(&self, name: &str, value: &str) -> std::io::Result<()> {
488        let mut c = Command::new(&self.bin)
489            .arg("--actor")
490            .arg(&self.actor)
491            .arg("put")
492            .arg(self.rref(name))
493            .stdin(Stdio::piped())
494            .stderr(Stdio::null())
495            .spawn()?;
496        if let Some(mut stdin) = c.stdin.take() {
497            stdin.write_all(value.as_bytes())?;
498        }
499        let status = c.wait()?;
500        if status.success() {
501            Ok(())
502        } else {
503            Err(std::io::Error::other(format!(
504                "cuora put {} exited {status}",
505                self.rref(name)
506            )))
507        }
508    }
509}

713    fn serve_transport_roundtrips_through_the_wire() {
714        // A `cuora serve` stub in the ndjson wire shape — the same
715        // fake `serve.rs` tests use, inlined here so this test pins
716        // the *Secrets* dispatch, not the codec.
717        let dir = tempfile::tempdir().unwrap();
718        let db = dir.path().join("db");
719        let script = format!(
720            "#!/bin/sh\n\
721             while IFS= read -r line; do\n\
722               op=$(printf '%s' \"$line\" | grep -o '\"op\":\"[a-z]*\"' | cut -d'\"' -f4)\n\
723               ref=$(printf '%s' \"$line\" | grep -o '\"ref\":\"[^\"]*\"' | cut -d'\"' -f4)\n\
724               val=$(printf '%s' \"$line\" | grep -o '\"value\":\"[^\"]*\"' | cut -d'\"' -f4)\n\
725               key=${{ref##*/}}\n\
726               if [ \"$op\" = put ]; then printf '%s' \"$val\" > '{db}-'\"$key\"; \\\n                 printf '{{\"id\":1,\"ok\":true,\"result\":{{}}}}\\n';\n\
727               elif [ \"$op\" = get ] && [ -f '{db}-'\"$key\" ]; then \\\n                 printf '{{\"id\":1,\"ok\":true,\"result\":{{\"value\":\"%s\"}}}}\\n' \"$(cat '{db}-'\"$key\")\";\n\
728               else printf '{{\"id\":1,\"ok\":false,\"error\":{{\"code\":\"not_found\",\"message\":\"x\"}}}}\\n'; fi\n\
729             done\n",
730            db = db.display(),
731        );
732        let bin = stub(dir.path(), &script);
733        let s = CuoraSecrets::new("demo")
734            .with_bin(bin)
735            .with_transport(CuoraTransport::Serve);
736        assert_eq!(s.get("MISSING"), None);
737        s.set("TOKEN", "sekret").unwrap();
738        assert_eq!(s.get("TOKEN").as_deref(), Some("sekret"));
739    }
